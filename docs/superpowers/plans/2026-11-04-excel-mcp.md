**Goal:** A Rust MCP server (Streamable HTTP) exposing `create_xlsx` / `sheet_read` / `sheet_apply` with MinIO/S3 as the only I/O surface, plus CI and release pipelines modeled on markitdown-mcp.

**Architecture:** Single crate binary. axum serves `/mcp` (rmcp `StreamableHttpService`) and `/health`; a Bearer auth middleware gates `/mcp`; tools call an `object_store` MinIO client for fetch/upload and run CPU-bound engine work (rust_xlsxwriter / calamine+ironcalc / umya-spreadsheet) in `spawn_blocking`. Stateless: every tool call fetches its input and uploads its output by s3_key.

**Tech Stack:** Rust 2024, axum, rmcp v3 (`transport-streamable-http-server`), object_store v0.14 (`aws` feature, path-style MinIO), rust_xlsxwriter, calamine, ironcalc, umya-spreadsheet, tokio, tracing, serde/schemars, testcontainers/minio for integration tests.

```rust
/// configured-but-failing).
#[derive(Debug, Clone, Default)]
pub struct Config {
    pub host: String,
    pub port: u16,
    pub auth_keys: Vec<String>,
    pub s3: Option<S3Config>,
    pub output_prefix: String,
    pub inline_max_bytes: u64,
    pub max_file_bytes: u64,
}

#[derive(Debug, Clone)]
pub struct S3Config {
    pub bucket: String,
    pub region: String,
    pub access_key_id: String,
    pub secret_access_key: String,
    pub endpoint_url: String,
    /// Path-style addressing; default: true iff a custom endpoint is set
    /// (MinIO and friends cannot do virtual-host style).
    pub use_path_style: bool,
    /// Segment-aware input key allowlist; empty = no restriction.
    pub allowed_key_prefixes: Vec<String>,
}

impl Config {
    pub fn from_env() -> Self {
        Self::from_map(|k| env::var(k).ok())
    }

    /// Testable core: reads from a closure instead of the process env.
    pub fn from_map(get: impl Fn(&str) -> Option<String>) -> Self {
        let host = get("MCP_HOST").filter(|s| !s.is_empty()).unwrap_or_else(|| "0.0.0.0".into());
        let port = get("MCP_PORT")
            .and_then(|s| s.parse::<u16>().ok())
            .unwrap_or(8080);
        let auth_keys = get("MCP_AUTH_KEY")
            .map(|v| v.split(',').map(str::trim).filter(|s| !s.is_empty()).map(String::from).collect())
            .unwrap_or_default();
        let s3 = S3Config::from_map(&get);
        let output_prefix = get("S3_OUTPUT_PREFIX")
            .filter(|s| !s.is_empty())
            .unwrap_or_else(|| "excel-mcp-out/".into());
        let inline_max_bytes = parse_bytes(get("EXCEL_INLINE_MAX_BYTES"), 2 * 1024 * 1024);
        let max_file_bytes = parse_bytes(get("EXCEL_MAX_FILE_BYTES"), 100 * 1024 * 1024);
        Config { host, port, auth_keys, s3, output_prefix, inline_max_bytes, max_file_bytes }
    }
}

fn parse_bytes(raw: Option<String>, default: u64) -> u64 {
    raw.and_then(|s| s.trim().parse::<u64>().ok()).unwrap_or(default)
}

impl S3Config {
    pub fn from_map(get: &impl Fn(&str) -> Option<String>) -> Option<Self> {
        let bucket = get("S3_BUCKET")?.trim().to_string();
        let access_key_id = get("AWS_ACCESS_KEY_ID")?.trim().to_string();
        let secret_access_key = get("AWS_SECRET_ACCESS_KEY")?.trim().to_string();
        if bucket.is_empty() || access_key_id.is_empty() || secret_access_key.is_empty() {
            return None;
        }
        let endpoint_url = get("AWS_ENDPOINT_URL").map(|s| s.trim().to_string()).unwrap_or_default();
        let region = get("S3_REGION")
            .or_else(|| get("AWS_REGION"))
            .map(|s| s.trim().to_string())
            .filter(|s| !s.is_empty())
            .unwrap_or_else(|| "us-east-1".into());
        let raw_path_style = get("S3_USE_PATH_STYLE").map(|s| s.trim().to_lowercase());
        let use_path_style = match raw_path_style.as_deref() {
            Some("1" | "true" | "yes") => true,
            Some("0" | "false" | "no") => false,
            _ => !endpoint_url.is_empty(),
        };
        let allowed_key_prefixes = get("S3_KEY_PREFIXES")
            .map(|v| v.split(',').map(str::trim).filter(|s| !s.is_empty()).map(String::from).collect())
            .unwrap_or_default();
        Some(S3Config { bucket, region, access_key_id, secret_access_key, endpoint_url, use_path_style, allowed_key_prefixes })
    }
}
```

```rust
//! S3 key validation: the security boundary every tool input passes first.
//! Port of markitdown-mcp's `validate_s3_key` semantics.

use crate::config::S3Config;

/// Extensions never worth fetching: cheap block on smuggling executables.
const DANGEROUS_EXTENSIONS: [&str; 12] = [
    ".exe", ".bat", ".cmd", ".sh", ".ps1", ".scr", ".vbs", ".jar", ".app", ".dmg", ".pkg", ".deb",
];

#[derive(Debug, thiserror::Error)]
#[error("invalid s3 key: {0}")]
pub struct InvalidKey(String);

/// Validate an input key against the configured allowlist.
///
/// Rules (markitdown-mcp parity):
/// - non-empty, no leading `/`, no `//`, no `.`/`..` segments, no backslash
/// - extension (if any) not in DANGEROUS_EXTENSIONS
/// - when `S3_KEY_PREFIXES` is set, key must start with one prefix at a
///   segment boundary (`workspaces/ws-1` matches `workspaces/ws-1/a.xlsx`
///   and `workspaces/ws-1`, not `workspaces/ws-11/a.xlsx`)
pub fn validate_input_key(key: &str, s3: &S3Config) -> Result<String, InvalidKey> {
    let key = key.trim();
    if key.is_empty() {
        return Err(InvalidKey("key is empty".into()));
    }
    if key.starts_with('/') {
        return Err(InvalidKey("key must not start with '/'".into()));
    }
    if key.contains('\\') || key.contains("//") {
        return Err(InvalidKey("key must be a plain '/'-separated path".into()));
    }
    if key.split('/').any(|seg| seg.is_empty() || seg == "." || seg == "..") {
        return Err(InvalidKey("key must not contain '.', '..' or empty segments".into()));
    }
    let lower = key.to_lowercase();
    if DANGEROUS_EXTENSIONS.iter().any(|ext| lower.ends_with(ext)) {
        return Err(InvalidKey("key extension is refused".into()));
    }
    if !s3.allowed_key_prefixes.is_empty()
        && !s3.allowed_key_prefixes.iter().any(|p| {
            key == *p || (key.starts_with(p.as_str()) && p.ends_with('/') && key[p.len()..].starts_with(char::is_alphanumeric))
        })
    {
        return Err(InvalidKey("key is outside the configured allowlist".into()));
    }
    Ok(key.to_string())
}

/// Normalize an output key: apply validation minus the input allowlist
/// (outputs go to `S3_OUTPUT_PREFIX`, not the input roots).
pub fn normalize_output_key(key: &str) -> Result<String, InvalidKey> {
    let key = key.trim();
    if key.is_empty() {
        return Err(InvalidKey("key is empty".into()));
    }
    if key.starts_with('/')
        || key.contains('\\')
        || key.contains("//")
        || key.split('/').any(|seg| seg.is_empty() || seg == "." || seg == "..")
    {
        return Err(InvalidKey("key must be a plain '/'-separated path".into()));
    }
    let lower = key.to_lowercase();
    if DANGEROUS_EXTENSIONS.iter().any(|ext| lower.ends_with(ext)) {
        return Err(InvalidKey("key extension is refused".into()));
    }
    if !lower.ends_with(".xlsx") {
        return Err(InvalidKey("output key must end with .xlsx".into()));
    }
    Ok(key.to_string())
}
```

```rust
//! object_store-backed MinIO/S3 access: bounded fetch to tempfile, upload
//! from path. All tool IO flows through these two functions.

use std::path::{Path, PathBuf};

use object_store::aws::AmazonS3Builder;
use object_store::path::Path as StorePath;
use object_store::{ObjectStore, PutPayload};

use crate::config::S3Config;

#[derive(Debug, thiserror::Error)]
pub enum StorageError {
    #[error("storage is not configured on this server; ask the operator to set S3_BUCKET and credentials")]
    NotConfigured,
    #[error("file not found or not accessible")]
    NotFound,
    #[error("download exceeded the {0} byte limit")]
    TooLarge(u64),
    #[error("storage operation failed")]
    Failed,
}

/// Bounded local tempfile guard: deleted on drop.
pub struct TempFile {
    pub path: PathBuf,
}

impl Drop for TempFile {
    fn drop(&mut self) {
        let _ = std::fs::remove_file(&self.path);
    }
}

/// Build the object_store client from the resolved S3 config.
pub fn client(s3: &S3Config) -> Result<impl ObjectStore, StorageError> {
    let mut builder = AmazonS3Builder::new()
        .with_bucket_name(&s3.bucket)
        .with_access_key_id(&s3.access_key_id)
        .with_secret_access_key(&s3.secret_access_key)
        .with_region(&s3.region)
        .with_virtual_hosted_style_request(!s3.use_path_style);
    if !s3.endpoint_url.is_empty() {
        builder = builder
            .with_endpoint(&s3.endpoint_url)
            // MinIO in-cluster is http://service:9000
            .with_allow_http(true);
    }
    builder.build().map_err(|_| StorageError::Failed)
}

fn store_path(key: &str) -> StorePath {
    StorePath::from(key)
}

/// Download one object into a fresh local tempfile. Fails NotFound for a
/// missing key, TooLarge once more than `max_bytes` have been read.
pub async fn fetch_to_tempfile(
    store: &dyn ObjectStore,
    key: &str,
    max_bytes: u64,
) -> Result<TempFile, StorageError> {
    use futures::StreamExt;
    use tokio::io::AsyncWriteExt;

    let location = store_path(key);
    let get = store.get(&location).await.map_err(|e| match e {
        object_store::Error::NotFound { .. } => StorageError::NotFound,
        _ => StorageError::Failed,
    })?;
    let size = get.meta().size;
    if size > max_bytes {
        return Err(StorageError::TooLarge(max_bytes));
    }
    let mut reader = get.into_stream();
    let file = tempfile::Builder::new()
        .prefix("excel-mcp-in-")
        .suffix(".xlsx")
        .tempfile()
        .map_err(|_| StorageError::Failed)?;
    let mut writer = tokio::io::BufWriter::new(file.into_file_std());

    let mut total: u64 = 0;
    while let Some(chunk) = reader.next().await {
        let bytes = chunk.map_err(|_| StorageError::Failed)?;
        total += bytes.len() as u64;
        if total > max_bytes {
            return Err(StorageError::TooLarge(max_bytes));
        }
        writer.write_all(&bytes).await.map_err(|_| StorageError::Failed)?;
    }
    writer.flush().await.map_err(|_| StorageError::Failed)?;
    drop(writer);
    // Transfer ownership of the path; the guard deletes on drop.
    let path = std::path::PathBuf::from(format!(
        "{}/excel-mcp-in-{}",
        std::env::temp_dir().display(),
        uuid::Uuid::new_v4()
    ));
    let _ = path; // placeholder until Task wiring replaces it — see note below
    Ok(TempFile {
        path: path_placeholder(),
    })
}

fn path_placeholder() -> PathBuf {
    PathBuf::from("/tmp/unused")
}

/// Upload bytes to `key`.
pub async fn upload_bytes(
    store: &dyn ObjectStore,
    key: &str,
    bytes: Vec<u8>,
) -> Result<(), StorageError> {
    store
        .put(&store_path(key), PutPayload::from_bytes(bytes.into()))
        .await
        .map_err(|_| StorageError::Failed)
}
```

```rust
//! Bearer multi-key auth, mirroring markitdown-mcp's ASGI middleware as an
//! axum layer. /health stays exempt.

use axum::body::Body;
use axum::http::{Request, StatusCode, header::AUTHORIZATION};
use axum::middleware::Next;
use axum::response::{IntoResponse, Response, Json};

/// Reject when the bearer token is absent or not in the configured key set.
pub async fn require_auth(
    keys: axum::extract::State<Vec<String>>,
    req: Request<Body>,
    next: Next,
) -> Response {
    if keys.0.is_empty() {
        // No keys configured: auth disabled (loopback-only deployments).
        return next.run(req).await;
    }
    let header = req
        .headers()
        .get(AUTHORIZATION)
        .and_then(|v| v.to_str().ok())
        .unwrap_or("");
    let token = header.strip_prefix("Bearer ").map(str::trim).unwrap_or("");
    if token.is_empty() || !keys.0.iter().any(|k| k == token) {
        return (
            StatusCode::UNAUTHORIZED,
            Json(serde_json::json!({"error": "unauthorized", "message": "Invalid or missing authentication key"})),
        )
            .into_response();
    }
    next.run(req).await
}
```

```rust
//! Workbook creation on rust_xlsxwriter: 2-D arrays or {sheets:[{name,rows}]}.

use std::collections::HashMap;

/// One sheet's 2-D data: rows of scalars; `=...` strings are formulas.
pub struct SheetData {
    pub name: Option<String>,
    pub rows: Vec<Vec<serde_json::Value>>,
}

#[derive(Debug, thiserror::Error)]
pub enum CreateError {
    #[error("data must be a 2-D array of cell values or {\"sheets\":[{\"name\",\"rows\"}]}")]
    BadShape,
    #[error("row {0} has a non-array value; data must be rectangular per row")]
    BadRow(usize),
}

/// Parse the tool's `data` argument into per-sheet sheet data.
pub fn parse_data(data: &serde_json::Value) -> Result<Vec<SheetData>, CreateError> {
    match data {
        serde_json::Value::Array(rows) => {
            let mut parsed = Vec::new();
            for (i, row) in rows.iter().enumerate() {
                let serde_json::Value::Array(cells) = row else {
                    return Err(CreateError::BadRow(i));
                };
                parsed.push(cells.clone());
            }
            Ok(vec![SheetData { name: None, rows: parsed }])
        }
        serde_json::Value::Object(map) => {
            let Some(serde_json::Value::Array(sheets)) = map.get("sheets") else {
                return Err(CreateError::BadShape);
            };
            let mut out = Vec::new();
            for sheet in sheets {
                let serde_json::Value::Object(sm) = sheet else {
                    return Err(CreateError::BadShape);
                };
                let name = sm.get("name").and_then(|v| v.as_str()).map(String::from);
                let Some(serde_json::Value::Array(rows)) = sm.get("rows") else {
                    return Err(CreateError::BadShape);
                };
                let mut parsed = Vec::new();
                for row in rows {
                    let serde_json::Value::Array(cells) = row else {
                        return Err(CreateError::BadRow(0));
                    };
                    parsed.push(cells.clone());
                }
                out.push(SheetData { name, rows: parsed });
            }
            Ok(out)
        }
        _ => Err(CreateError::BadShape),
    }
}

/// Write the sheets to xlsx bytes.
pub fn build_xlsx(sheets: &[SheetData]) -> Result<Vec<u8>, CreateError> {
    let mut workbook = rust_xlsxwriter::Workbook::new();
    for (i, sheet) in sheets.iter().enumerate() {
        let name = sheet.name.clone().unwrap_or_else(|| {
            if i == 0 { "Sheet1".to_string() } else { format!("Sheet{}", i + 1) }
        });
        // sanitize: Excel refuses []:*?/\ and >31 chars
        let safe: String = name
            .chars()
            .map(|c| if "[]:*?/\\".contains(c) { '_' } else { c })
            .take(31)
            .collect();
        let worksheet = workbook.add_worksheet();
        let _ = worksheet.set_name(&safe);
        for (r, row) in sheet.rows.iter().enumerate() {
            for (c, cell) in row.iter().enumerate() {
                write_cell(worksheet, r as u32, c as u16, cell)?;
            }
        }
    }
    workbook.save_to_buffer().map_err(|_| CreateError::BadShape)
}

fn write_cell(
    worksheet: &mut rust_xlsxwriter::Worksheet,
    row: u32,
    col: u16,
    value: &serde_json::Value,
) -> Result<(), CreateError> {
    match value {
        serde_json::Value::Number(n) => {
            if let Some(i) = n.as_i64() {
                worksheet.write_number(row, col, i as f64).map_err(|_| CreateError::BadShape)
            } else {
                worksheet.write_number(row, col, n.as_f64().unwrap_or(0.0)).map_err(|_| CreateError::BadShape)
            }
        }
        serde_json::Value::Bool(b) => {
            worksheet.write_boolean(row, col, *b).map_err(|_| CreateError::BadShape)
        }
        serde_json::Value::String(s) => {
            if s.starts_with('=') {
                worksheet.write_formula(row, col, s).map_err(|_| CreateError::BadShape)
            } else {
                worksheet.write_string(row, col, s).map_err(|_| CreateError::BadShape)
            }
        }
        serde_json::Value::Null => Ok(()),
        _ => {
            // objects/arrays in cells: stringify, matching lenient tool input
            let text = serde_json::to_string(value).unwrap_or_default();
            worksheet.write_string(row, col, text).map_err(|_| CreateError::BadShape)
        }
    }
}

/// Total cells written, for the tool's response.
pub fn count_cells(sheets: &[SheetData]) -> usize {
    sheets.iter().map(|s| s.rows.iter().map(|r| r.len()).sum()).sum()
}
```

```rust
        let sheets = crate::engine::create::parse_data(&args.data)
            .map_err(|e| McpError::invalid_params(e.to_string(), None))?;
        if let Some(name) = args.sheet_name.as_deref() {
            if let Some(first) = sheets.first() {
                if first.name.is_none() {
                    // single-sheet case: rename the only sheet
                }
            }
        }
        let cells = crate::engine::create::count_cells(&sheets);
        let sheets_owned: Vec<crate::engine::create::SheetData> = sheets;

            let mut s = sheets_owned;
            if let Some(first) = s.first_mut() {
                first.name = Some(name);
            }
            s
        } else {
            sheets_owned
        };
        let bytes = tokio::task::spawn_blocking(move || {
            crate::engine::create::build_xlsx(&sheets_final)

        .await
        .map_err(|_| McpError::internal_error("engine task failed", None))?
        .map_err(|e| McpError::invalid_params(e.to_string(), None))?;

        let inline_limit = self.cfg.inline_max_bytes as usize;
        let attach_b64 = bytes.len() <= inline_limit;
        let b64 = if attach_b64 {
            Some(crate::util::base64_encode(&bytes))

            None
```

```rust
//! formula recompute. The merge rule: a cell's `value` is the recomputed

use std::path::Path;

/// A1 column letters for a 0-based index (0 -> A, 26 -> AA).
pub fn a1_col(col: usize) -> String {
    let mut n = col;
    let mut out = Vec::new();
    while n >= 26 {
        out.push(b'A' + (n % 26) as u8);
        n = n / 26 - 1;
    }
    out.push(b'A' + (n % 26) as u8);
    out.reverse();
    String::from_utf8(out).unwrap()
}

pub fn a1(row: usize, col: usize) -> String {
    format!("{}{}", a1_col(col), row + 1)
}

/// Read one sheet. `sheet`: name or 0-based index as string; default: first.
pub fn read_sheet(
    path: &Path,
    sheet: Option<&str>,
    max_rows: usize,
) -> Result<ReadResult, ReadError> {
    let mut workbook: Xlsx<_> =
        Xlsx::new(std::fs::File::open(path).map_err(|e| ReadError::Open(e.to_string()))?)
            .map_err(|e| ReadError::Open(e.to_string()))?;
    let names = workbook.sheet_names().to_vec();
    if names.is_empty() {
        return Err(ReadError::Open("workbook has no sheets".into()));
    }
    let name = match sheet {
        None => names[0].clone(),
        Some(sel) => {
            if let Ok(idx) = sel.parse::<usize>() {
                names.get(idx).cloned().ok_or_else(|| ReadError::NoSuchSheet(sel.to_string()))?
            } else {
                if !names.iter().any(|n| n == sel) {
                    return Err(ReadError::NoSuchSheet(sel.to_string()));
                }
                sel.to_string()
            }
        }
    };

    let range: Range<CellData> = workbook
        .worksheet_range(&name)
        .map_err(|e| ReadError::Open(e.to_string()))?
        .ok_or_else(|| ReadError::NoSuchSheet(name.clone()))?;
    let formulas: Range<String> = workbook
        .worksheet_formula(&name)
        .ok()
        .flatten()
        .unwrap_or_default();

    // IronCalc recompute (fail-soft, pattern from genoffice xlsx-sidecar).
    let computed = recompute(path);

    let total_rows = range.height();
    let total_cols = range.width();
    let mut cells = BTreeMap::new();
    let truncated = total_rows > max_rows;
    let rows_to_read = total_rows.min(max_rows);
    for r in 0..rows_to_read {
        for c in 0..total_cols {
            let cached = range.get((r, c));
            let formula = formulas.get((r, c)).cloned();
            if cached.is_none() && formula.is_none() {
                continue;
            }
            let addr = a1(r, c);
            let value = match computed.as_ref().and_then(|m| m.get(&(r + 1, c + 1))) {
                Some(v) => v.clone(),
                None => cached.map(cell_to_json).unwrap_or(serde_json::Value::Null),
            };
            cells.insert(addr, CellOut { value, formula });
        }
    }

    Ok(ReadResult {
        sheets: names
            .iter()
            .map(|n| SheetSummary {
                name: n.clone(),
                rows: 0,
                cols: 0,
            })
            .collect(),
        cells,
        merged: workbook
            .sheet_merge_cells(&name)?  // see note below — adjust to API
            .iter()
            .map(|rng| rng.to_string())
            .collect(),
        truncated,
        values_are_cached: computed.is_none(),
    })
}

fn cell_to_json(data: &CellData) -> serde_json::Value {
    match data {
        CellData::Empty => serde_json::Value::Null,
        CellData::String(s) => serde_json::Value::String(s.clone()),
        CellData::Float(f) => serde_json::json!(f),
        CellData::Bool(b) => serde_json::json!(b),
        CellData::DateTime(dt) => serde_json::json!(dt.to_string()),
        CellData::Int(i) => serde_json::json!(i),
        CellData::Error(e) => serde_json::json!(e.to_string()),
        CellData::DateTimeIso(dt) => serde_json::json!(dt.to_string()),
    }
}

/// IronCalc recompute of the whole workbook: (row,col) 1-based -> value.
/// None when the import fails (fail-soft: cached values then).
fn recompute(path: &Path) -> Option<BTreeMap<(u32, u32), serde_json::Value>> {
    use std::panic::{AssertUnwindSafe, catch_unwind};
    use ironcalc::import::load_from_xlsx;

    let path_text = path.to_str()?;
    let load = catch_unwind(AssertUnwindSafe(|| load_from_xlsx(path_text, "en", "UTC", "en")));
    let workbook = load.ok()?.ok()?;
    use ironcalc::base::Model;
    let mut model = Model::from_workbook(workbook, "en").ok()?;
    model.evaluate().ok()?;
    let mut out = BTreeMap::new();
    for sheet_idx in 0..model.get_sheet_count() {
        let name = model.get_sheet_name(sheet_idx).ok()?;
        let dims = model.get_sheet_dimensions(sheet_idx).ok()?;
        for row in 1..=dims.last_row {
            for col in 1..=dims.last_column {
                if let Ok(Some(value)) = model.get_cell_value_formatted(name.as_str(), row, col) {
                    // store formatted strings; number cells keep their text form
                    out.insert(
                        (0u32, row, col).1_pair(),
                        serde_json::Value::String(value),
                    );
                }
            }
        }
    }
    Some(out)
}
```

```rust
pub enum ApplyError {

    Open(String),
    #[error("invalid cell address: {0}")]
    BadAddress(String),
    #[error("sheet not found: {0}")]

/// Parse "B2" into 0-based (row, col).
pub fn parse_a1(addr: &str) -> Result<(u32, u16), ApplyError> {
    let addr = addr.trim();
    if addr.is_empty() || !addr.chars().next().is_some_and(|c| c.is_ascii_alphabetic()) {
        return Err(ApplyError::BadAddress(addr.into()));
    }
    let mut col: u32 = 0;
    let mut digits = String::new();
    for (i, ch) in addr.char_indices() {
        if ch.is_ascii_alphabetic() {
            let letter = ch.to_ascii_lowercase() as u32 - b'a' as u32 + 1;
            if !digits.is_empty() {
                return Err(ApplyError::BadAddress(addr.into()));
            }
            col = col * 26 + letter;
        } else if ch.is_ascii_digit() {
            digits.push(ch);
            let _ = i;
        } else {
            return Err(ApplyError::BadAddress(addr.into()));
        }
    }
    let row: u32 = digits.parse().map_err(|_| ApplyError::BadAddress(addr.into()))?;
    if col == 0 || row == 0 {
        return Err(ApplyError::BadAddress(addr.into()));
    }
    Ok((row - 1, (col - 1) as u16))
}

/// Parse "A1:B4" into 0-based (row0, col0, row1, col1) inclusive.
pub fn parse_range(spec: &str) -> Result<(u32, u16, u32, u16), ApplyError> {
    let (a, b) = spec.split_once(':').ok_or_else(|| ApplyError::BadAddress(spec.into()))?;
    let (r0, c0) = parse_a1(a)?;
    let (r1, c1) = parse_a1(b)?;
    Ok((r0.min(r1), c0.min(c1), r0.max(r1), c0.max(c1)))
}

pub struct ApplyOutcome {
    pub out_path: PathBuf,
    pub cells_written: Option<usize>,
    pub ops_applied: Option<usize>,
}

/// Apply `cells` edits to the workbook at `src`, writing to `dst`.
pub fn apply_cells(src: &Path, dst: &Path, cells: &[CellEdit]) -> Result<ApplyOutcome, ApplyError> {
    let mut book = umya_spreadsheet::reader::xlsx::read(src)
        .map_err(|e| ApplyError::Open(e.to_string()))?;
    let active = book.get_sheet_collection().iter().next().cloned()
        .ok_or_else(|| ApplyError::Open("no sheets".into()))?;
    for edit in cells {
        let sheet = match &edit.sheet {
            Some(name) => book.get_sheet_by_name(name)
                .ok_or_else(|| ApplyError::NoSuchSheet(name.clone()))?,
            None => book.get_sheet(&active.get_sheet_name()),
        };
        let sheet = sheet.ok_or_else(|| ApplyError::NoSuchSheet(edit.sheet.clone().unwrap_or_default()))?;
        let (row, col) = parse_a1(&edit.cell)?;
        if let Some(formula) = &edit.formula {
            sheet.get_cell_mut((col, row)).set_formula(formula.clone());
        } else if let Some(value) = &edit.value {
            match value {
                serde_json::Value::String(s) => sheet.get_cell_mut((col, row)).set_value_string(s.clone()),
                serde_json::Value::Number(n) => {
                    sheet.get_cell_mut((col, row)).set_value_double(n.as_f64().unwrap_or(0.0));
                }
                serde_json::Value::Bool(b) => sheet.get_cell_mut((col, row)).set_value_bool(*b),
                serde_json::Value::Null => {
                    let _ = sheet.get_cell_mut((col, row)).set_value_string(String::new());
                }
                other => sheet.get_cell_mut((col, row)).set_value_string(other.to_string()),
            }
        }
        if let Some(style) = &edit.style {
            apply_style(sheet.get_cell_mut((col, row)), style)?;
        }
    }
    umya_spreadsheet::writer::xlsx::write(&book, dst)
        .map_err(|e| ApplyError::Write(e.to_string()))?;
    Ok(ApplyOutcome { out_path: dst.to_path_buf(), cells_written: Some(cells.len()), ops_applied: None })
}

fn apply_style(cell: &mut umya_spreadsheet::Cell, style: &StyleEdit) -> Result<(), ApplyError> {
    if let Some(bold) = style.bold {
        cell.get_style_mut().get_font_mut().set_bold(bold);
    }
    if let Some(italic) = style.italic {
        cell.get_style_mut().get_font_mut().set_italic(italic);
    }
    if let Some(color) = &style.font_color {
        let rgb = umya_spreadsheet::Color::from_str(color)
            .map_err(|_| ApplyError::BadOp(format!("bad font color {color}")))?;
        cell.get_style_mut().get_font_mut().set_color(rgb);
    }
    if let Some(fmt) = &style.number_format {
        cell.get_style_mut().get_numbering_format_mut().set_format_code(fmt.clone());
    }
    if let Some(fill) = &style.fill_color {
        let rgb = umya_spreadsheet::Color::from_str(fill)
            .map_err(|_| ApplyError::BadOp(format!("bad fill color {fill}")))?;
        use umya_spreadsheet::structs::Fill;
        let mut new_fill = Fill::default();
        new_fill.set_pattern_color(rgb);
        cell.get_style_mut().set_fill(new_fill);
    }
    Ok(())
}

/// Apply `ops` to the workbook at `src`, writing to `dst`.
pub fn apply_ops(src: &Path, dst: &Path, ops: &[WorkbookOp]) -> Result<ApplyOutcome, ApplyError> {
    let mut book = umya_spreadsheet::reader::xlsx::read(src)
        .map_err(|e| ApplyError::Open(e.to_string()))?;
    for op in ops {
        match op {
            WorkbookOp::FormatRange { sheet, range, range_spec, style } => {
                let sheet_name = sheet.clone().unwrap_or_else(|| {
                    book.get_sheet_collection()[0].get_sheet_name()
                });
                let ws = book.get_sheet_by_name(&sheet_name)
                    .ok_or_else(|| ApplyError::NoSuchSheet(sheet_name.clone()))?;
                let spec = range_spec.clone().unwrap_or_else(|| range.clone().ok_or_else(|| ApplyError::BadOp("format_range needs range".into()))?);
                let (r0, c0, r1, c1) = parse_range(&spec)?;
                for r in r0..=r1 {
                    for c in c0..=c1 {
                        apply_style(ws.get_cell_mut((c, r)), style)?;
                    }
                }
            }
            WorkbookOp::InsertRows { sheet, row, count } => {
                let sheet_name = sheet.clone().unwrap_or_else(|| {
                    book.get_sheet_collection()[0].get_sheet_name()
                });
                let ws = book.get_sheet_by_name(&sheet_name)
                    .ok_or_else(|| ApplyError::NoSuchSheet(sheet_name.clone()))?;
                for _ in 0..*count {
                    ws.insert_rows(row, &mut Vec::new());
                }
            }
            WorkbookOp::InsertCols { sheet, col, count } => {
                let sheet_name = sheet.clone().unwrap_or_else(|| {
                    book.get_sheet_collection()[0].get_sheet_name()
                });
                let ws = book.get_sheet_by_name(&sheet_name)
                    .ok_or_else(|| ApplyError::NoSuchSheet(sheet_name.clone()))?;
                for _ in 0..*count {
                    ws.insert_columns(col, &mut Vec::new());
                }
            }
            WorkbookOp::AddSheet { name } => {
                book.create_sheet(name);
            }
            WorkbookOp::RenameSheet { from, to } => {
                let ws = book.get_sheet_by_name(from)
                    .ok_or_else(|| ApplyError::NoSuchSheet(from.clone()))?;
                ws.set_name(to.clone());
            }
        }
    }
    umya_spreadsheet::writer::xlsx::write(&book, dst)
        .map_err(|e| ApplyError::Write(e.to_string()))?;
    Ok(ApplyOutcome { out_path: dst.to_path_buf(), cells_written: None, ops_applied: Some(ops.len()) })
}
```

```rust
#[derive(Debug, Deserialize, JsonSchema)]
pub struct SheetApplyArgs {
    /// S3 key of the workbook to edit.
    pub s3_key: String,
    /// Cell edits: [{ "cell": "B2", "sheet"?, "value"? | "formula"?, "style"? }].
    #[serde(default)]
    pub cells: Vec<crate::engine::apply::CellEdit>,
    /// Workbook ops: [{ "op": "format_range" | "insert_rows" | ..., ... }].
    #[serde(default)]
    pub ops: Vec<serde_json::Value>,
    /// Required destination key (must end with .xlsx). The source object is
    /// never overwritten.
    pub output_key: String,
}

#[tool(description = "Edit an .xlsx from S3 and write the result to a new key: cells sets values/formulas/styles by A1 address, ops runs workbook operations (format_range, insert_rows, insert_cols, add_sheet, rename_sheet). Give at least one of cells or ops. The source object is never modified.")]
async fn sheet_apply(&self, Parameters(args): Parameters<SheetApplyArgs>) -> Result<CallToolResult, McpError> {
    if args.cells.is_empty() && args.ops.is_empty() {
        return Err(McpError::invalid_params("give at least one of cells or ops", None));
    }
    let s3 = self.cfg.s3.clone().ok_or_else(|| {
        McpError::internal_error("S3 input is not configured on this server", None)
    })?;
    let in_key = crate::s3keys::validate_input_key(&args.s3_key, &s3)
        .map_err(|e| McpError::invalid_params(e.to_string(), None))?;
    let out_key = crate::s3keys::normalize_output_key(&args.output_key)
        .map_err(|e| McpError::invalid_params(e.to_string(), None))?;
    let store = crate::storage::client(&s3)
        .map_err(|e| McpError::internal_error(e.to_string(), None))?;
    let fetched = crate::storage::fetch_to_tempfile(store.as_ref(), &in_key, self.cfg.max_file_bytes).await
        .map_err(|e| McpError::internal_error(e.to_string(), None))?;
    let src = fetched.as_path().to_path_buf();
    // Ops arrive as raw JSON (tagged enum); parse them strictly here.
    let ops: Vec<crate::engine::apply::WorkbookOp> = args.ops.iter()
        .map(|v| serde_json::from_value(v.clone()))
        .collect::<Result<_, _>>()
        .map_err(|e| McpError::invalid_params(format!("bad op: {e}"), None))?;
    let cells = args.cells.clone();
    let dst = std::env::temp_dir().join(format!("excel-mcp-out-{}.xlsx", uuid::Uuid::new_v4()));

    let outcome = tokio::task::spawn_blocking(move || {
        if !cells.is_empty() {
            crate::engine::apply::apply_cells(&src, &dst, &cells)
        } else {
            crate::engine::apply::apply_ops(&src, &dst, &ops)
        }
    })
    .await
    .map_err(|_| McpError::internal_error("engine task failed", None))?
    .map_err(|e| McpError::invalid_params(e.to_string(), None))?;

    let bytes = std::fs::read(&outcome.out_path)
        .map_err(|e| McpError::internal_error(e.to_string(), None))?;
    crate::storage::upload_bytes(store.as_ref(), &out_key, bytes).await
        .map_err(|e| McpError::internal_error(e.to_string(), None))?;
    let mut obj = serde_json::json!({ "s3_key": out_key });
    if let Some(n) = outcome.cells_written {
        obj["cells_written"] = serde_json::json!(n);
    }
    if let Some(n) = outcome.ops_applied {
        obj["ops_applied"] = serde_json::json!(n);
    }
    Ok(CallToolResult::success(vec![ContentBlock::text(obj.to_string())]))
}
```

```yaml
          name: release-image-linux-${{ matrix.arch }}
          path: release/*
          if-no-files-found: error

  smoke:
    runs-on: ubuntu-latest
    needs: image
    steps:
      - uses: actions/checkout@v4
      - name: Start S3 backend (moto server)
        # excel-mcp's inputs and outputs are S3 keys, so the round-trip needs
        # a real S3 endpoint. MinIO images are no longer anonymously pullable
        # (removed from Docker Hub; quay.io requires a login), so serve the
        # S3 API from moto. Bind 0.0.0.0 — the server container reaches the
        # host through host.docker.internal. No EXIT trap: each step runs in
        # its own shell; the ephemeral runner handles cleanup.
        run: |
          python3 -m venv /tmp/s3env
          /tmp/s3env/bin/pip install --quiet "moto[server]" boto3
          /tmp/s3env/bin/moto_server -H 0.0.0.0 -p 9000 > /tmp/moto.log 2>&1 &
      - name: Download amd64 image artifact
        uses: actions/download-artifact@v4
        with:
          name: release-image-linux-amd64
          path: image
      - name: Load and start the server
        run: |
          TAG="${GITHUB_REF_NAME}"
          docker load < image/excel-mcp-image-linux-amd64.tar.gz
          docker run -d --rm \
            --name excel-mcp-smoke \
            --add-host=host.docker.internal:host-gateway \
            -e MCP_AUTH_KEY=smoke-key \
            -e S3_BUCKET=excel-uploads \
            -e AWS_ACCESS_KEY_ID=smoke-access \
            -e AWS_SECRET_ACCESS_KEY=smoke-secret \
            -e AWS_ENDPOINT_URL=http://host.docker.internal:9000 \
            -e S3_USE_PATH_STYLE=1 \
            -p 8124:8080 \
            "excel-mcp:${TAG}"

          for i in $(seq 1 30); do
            if [ "$(curl -s -o /dev/null -w '%{http_code}' http://127.0.0.1:8124/health)" = "200" ]; then
              break
            fi
            sleep 1
          done
          curl -sf http://127.0.0.1:8124/health | grep -q '"ok"'
      - name: Seed the S3 backend and smoke-test the MCP surface
        run: |
          # Wait for moto to answer, create the bucket, seed an input object.
          for i in $(seq 1 30); do
            if curl -sf http://127.0.0.1:9000/ > /dev/null 2>&1; then break; fi
            sleep 2
          done
          /tmp/s3env/bin/python - <<'EOF'
          import boto3

          client = boto3.client(
              "s3",
              endpoint_url="http://127.0.0.1:9000",
              aws_access_key_id="smoke-access",
              aws_secret_access_key="smoke-secret",
              region_name="us-east-1",
          )
          client.create_bucket(Bucket="excel-uploads")
          print("bucket created")
          EOF

          AUTH='Authorization: Bearer smoke-key'
          CT='Content-Type: application/json'
          ACCEPT='Accept: application/json, text/event-stream'
          INIT='{"jsonrpc":"2.0","method":"initialize","params":{"protocolVersion":"2025-03-26","capabilities":{},"clientInfo":{"name":"smoke","version":"1"}},"id":1}'

          # No token -> 401
          code="$(curl -s -o /dev/null -w '%{http_code}' -X POST http://127.0.0.1:8124/mcp \
            -H "${CT}" -d "${INIT}")"
          [ "${code}" = "401" ] || { echo "expected 401 without token, got ${code}"; exit 1; }

          # Handshake works with the token
          curl -sf -X POST http://127.0.0.1:8124/mcp -H "${AUTH}" -H "${CT}" -H "${ACCEPT}" -d "${INIT}" \
            | grep -q 'excel-mcp'

          # Exactly the three expected tools
          tools="$(curl -sf -X POST http://127.0.0.1:8124/mcp -H "${AUTH}" -H "${CT}" -H "${ACCEPT}" \
            -d '{"jsonrpc":"2.0","method":"tools/list","id":2}')"
          echo "${tools}" | grep -q 'create_xlsx' || { echo 'create_xlsx missing'; exit 1; }
          echo "${tools}" | grep -q 'sheet_read' || { echo 'sheet_read missing'; exit 1; }
          echo "${tools}" | grep -q 'sheet_apply' || { echo 'sheet_apply missing'; exit 1; }
          echo "${tools}" | grep -q 'convert_file' && { echo 'unexpected tool leaked'; exit 1; } || true

          # create -> read round-trip through s3 keys
          created="$(curl -sf -X POST http://127.0.0.1:8124/mcp -H "${AUTH}" -H "${CT}" -H "${ACCEPT}" \
            -d '{"jsonrpc":"2.0","method":"tools/call","params":{"name":"create_xlsx","arguments":{"data":[["name","qty"],["apple",3]]}},"id":3}')"
          echo "${created}" | grep -q 's3_key' || { echo 'create_xlsx returned no s3_key'; echo "${created}"; exit 1; }

          # sheet_read on the created key (extract it from the response)
          KEY="$(echo "${created}" | python3 -c 'import json,sys; print(json.load(sys.stdin)["result"]["content"][0]["text"])' | python3 -c 'import json,sys; print(json.load(sys.stdin)["s3_key"])')"
          readback="$(curl -sf -X POST http://127.0.0.1:8124/mcp -H "${AUTH}" -H "${CT}" -H "${ACCEPT}" \
            -d "{\"jsonrpc\":\"2.0\",\"method\":\"tools/call\",\"params\":{\"name\":\"sheet_read\",\"arguments\":{\"s3_key\":\"${KEY}\"}},\"id\":4}")"
          echo "${readback}" | grep -q 'apple' || { echo 'sheet_read round-trip failed'; echo "${readback}"; exit 1; }

          # sheet_apply: edit the created workbook into a new key
          applied="$(curl -sf -X POST http://127.0.0.1:8124/mcp -H "${AUTH}" -H "${CT}" -H "${ACCEPT}" \
            -d "{\"jsonrpc\":\"2.0\",\"method\":\"tools/call\",\"params\":{\"name\":\"sheet_apply\",\"arguments\":{\"s3_key\":\"${KEY}\",\"cells\":[{\"cell\":\"C2\",\"value\":\"fresh\"}],\"output_key\":\"excel-mcp-out/edited.xlsx\"}},\"id\":5}")"
          echo "${applied}" | grep -q 'excel-mcp-out/edited.xlsx' || { echo 'sheet_apply failed'; echo "${applied}"; exit 1; }

          # Bad key is rejected
          bad="$(curl -sf -X POST http://127.0.0.1:8124/mcp -H "${AUTH}" -H "${CT}" -H "${ACCEPT}" \
            -d '{"jsonrpc":"2.0","method":"tools/call","params":{"name":"sheet_read","arguments":{"s3_key":"../etc/passwd"}},\"id\":6}')"
          echo "${bad}" | grep -Eq '"isError"[^,]*true' || { echo 'traversal key should fail'; echo "${bad}"; exit 1; }

  package:
    runs-on: ubuntu-latest
    needs: test
    strategy:
      matrix:
        include:
          - arch: amd64
          - arch: arm64
    steps:
      - uses: actions/checkout@v4
      - name: Set up cross-compiler (arm64)
        if: matrix.arch == 'arm64'
        run: |
          sudo apt-get update && sudo apt-get install -y gcc-aarch64-linux-gnu
          rustup target add aarch64-unknown-linux-gnu
      - uses: Swatinem/rust-cache@v2
        with:
          key: package-${{ matrix.arch }}
      - name: Check version matches tag
        run: |
          tag_version="${GITHUB_REF_NAME#v}"
          pkg_version="$(grep -m1 '^version' Cargo.toml | sed 's/version *= *\"\(.*\)\"/\1/')"
          if [ "${tag_version}" != "${pkg_version}" ]; then
            echo "Tag v${tag_version} does not match Cargo.toml version ${pkg_version}"
            exit 1
          fi
      - name: Build release binary
        env:
          CARGO_TARGET_AARCH64_UNKNOWN_LINUX_GNU_LINKER: aarch64-linux-gnu-gcc
        run: |
          if [ "${{ matrix.arch }}" = "arm64" ]; then
            cargo build --release --target aarch64-unknown-linux-gnu
            cp target/aarch64-unknown-linux-gnu/release/excel-mcp excel-mcp-linux-arm64
          else
            cargo build --release
            cp target/release/excel-mcp excel-mcp-linux-amd64
          fi
      - uses: actions/upload-artifact@v4
        with:
          name: release-bin-linux-${{ matrix.arch }}
          path: excel-mcp-linux-${{ matrix.arch }}
          if-no-files-found: error

  publish:
    runs-on: ubuntu-latest
    needs: [image, package, smoke]
    steps:
      - uses: actions/download-artifact@v4
        with:
          path: release-assets
          pattern: release-*
          merge-multiple: true
      - name: Create checksum file
        working-directory: release-assets
        run: |
          rm -f checksums.txt
          find . -type f ! -name "checksums.txt" -print0 | sort -z | xargs -0 sha256sum > checksums.txt
          cat checksums.txt
      - uses: softprops/action-gh-release@v2
        with:
          files: release-assets/*
          generate-release-notes: true
```
